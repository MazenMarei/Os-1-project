                    }

                        cmd = inputPart1;

                    }
                }

                if (linux_cmd)

                    if (args)
                        snprintf(full_command, sizeof(full_command), "%s %s", linux_cmd, args);
                    else
                        snprintf(full_command, sizeof(full_command), "%s", linux_cmd);

                    system(full_command);
                }
                else

                    printf("Unrecognized DOS command: %s\n", cmd);
